```typescript
export interface QualityStats {
  usageCount: number       // จำนวน attempt
  reportRate: number       // reports / attempts
  goodRatingRatio: number  // 'good' / total ratings
  discrimination: number   // คนเก่งทำถูกมากกว่าคนอ่อน (point-biserial-ish, -1..1)
  rd: number               // rating deviation ปัจจุบัน
}

// เงื่อนไข AND (threshold = tune §9)
const TH = { usage: 50, reportRate: 0.05, goodRating: 0.7, discrimination: 0.2, rd: 100 }

export function isPromotable(s: QualityStats): { ok: boolean; reason?: string } {
  if (s.usageCount < TH.usage) return { ok: false, reason: 'usage ไม่พอ' }
  if (s.reportRate >= TH.reportRate) return { ok: false, reason: 'report เยอะ' }
  if (s.goodRatingRatio < TH.goodRating) return { ok: false, reason: 'rating ไม่ดีพอ' }
  if (s.discrimination < TH.discrimination) return { ok: false, reason: 'discrimination ต่ำ' }
  if (s.rd >= TH.rd) return { ok: false, reason: 'RD ยังสูง (ไม่ converge)' }
  return { ok: true }
}
```

```sql
-- candidate flag: ผ่าน auto-conditions แต่ยัง human spot-check (cold start)
alter table questions add column if not exists promote_candidate boolean not null default false;

-- job: คำนวณ stats ต่อ question → set promote_candidate (mock_eligible set โดยคน/auto ตาม phase)
create or replace function flag_promote_candidates() returns void language plpgsql as $$
begin
  update questions q set promote_candidate = true
  where q.confidence_score >= 70 and not q.mock_eligible and q.status = 'active'
    and q.rating_deviation < 100
    and (select count(*) from user_attempts a where a.question_id = q.id) >= 50
    and coalesce((select count(*)::float / nullif(count(*) over(),0) from reports r where r.question_id = q.id), 0) < 0.05;
end; $$;
-- cold start: คนตรวจ candidate → set mock_eligible=true, confidence_score=90
-- เมื่อ auto พิสูจน์แม่น: ใช้ trigger ตั้ง mock_eligible อัตโนมัติจาก promote_candidate
```

- [ ] **Step 2: implement**

`src/lib/mock/assembly.ts`:
```typescript
export interface EligibleQ { id: string; topic: string; mock_eligible: boolean }

export function assembleMock(pool: EligibleQ[], count: number): EligibleQ[] {
  const eligible = pool.filter((q) => q.mock_eligible)
  if (eligible.length < count) throw new Error('โจทย์ 90+ ไม่พอสำหรับ mock')
  // กระจาย topic: round-robin ตาม topic แล้วตัด count
  const byTopic = new Map<string, EligibleQ[]>()
  for (const q of eligible) {
    if (!byTopic.has(q.topic)) byTopic.set(q.topic, [])
    byTopic.get(q.topic)!.push(q)
  }
  const result: EligibleQ[] = []
  const topics = [...byTopic.keys()]
  let i = 0
  while (result.length < count) {
    const t = topics[i % topics.length]
    const bucket = byTopic.get(t)!
    if (bucket.length) result.push(bucket.shift()!)
    i++
  }
  return result
}
```

`src/lib/mock/scoring.ts`:
```typescript
export interface MockAnswer { topic: string; correct: boolean }
export interface Cutoff { min_score: number; median_score: number }

export function scoreMock(answers: MockAnswer[]): number {
  if (!answers.length) return 0
  return (answers.filter((a) => a.correct).length / answers.length) * 100
}

export function admitChance(score: number, c: Cutoff): 'สูง' | 'ปานกลาง' | 'ต่ำ' {
  if (score >= c.median_score) return 'สูง'
  if (score >= c.min_score) return 'ปานกลาง'
  return 'ต่ำ'
}

export function weakTopics(answers: MockAnswer[], threshold: number): string[] {
  const byTopic = new Map<string, { correct: number; total: number }>()
  for (const a of answers) {
    const t = byTopic.get(a.topic) ?? { correct: 0, total: 0 }
    t.total++; if (a.correct) t.correct++
    byTopic.set(a.topic, t)
  }
  return [...byTopic.entries()]
    .filter(([, v]) => v.correct / v.total < threshold)
    .map(([t]) => t)
}
```

```typescript
import { featuresFor } from '@/lib/billing/gating'
import { assembleMock } from '@/lib/mock/assembly'
import { scoreMock, admitChance, weakTopics } from '@/lib/mock/scoring'

export async function startMock(subject: string, count: number) {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return { error: 'unauthorized' }
  const { data: sub } = await supabase.from('subscriptions').select('status,tier').eq('user_id', user.id).single()
  if (!featuresFor(sub!).mock) return { error: 'mock เฉพาะสมาชิก (อัปเกรด)' }

  const { data: pool } = await supabase.from('questions')
    .select('id, topic, mock_eligible').eq('subject', subject).eq('mock_eligible', true).eq('status', 'active')
  const set = assembleMock((pool ?? []).map((q) => ({ ...q, mock_eligible: true })), count)
  return { ok: true, questionIds: set.map((q) => q.id) }
}

export async function submitMock(input: {
  subject: string; answers: { topic: string; questionId: string; correct: boolean }[]
}) {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return { error: 'unauthorized' }

  const score = scoreMock(input.answers)
  const weak = weakTopics(input.answers, 0.5)

  // โอกาสติดเทียบคณะเป้าหมาย
  const { data: goal } = await supabase.from('goals').select('target_faculty').eq('user_id', user.id).single()
  let chance: string | null = null
  if (goal?.target_faculty) {
    const { data: cutoff } = await supabase.from('faculty_cutoffs')
      .select('min_score, median_score').eq('faculty', goal.target_faculty).single()
    if (cutoff) chance = admitChance(score, cutoff)
  }

  // บันทึก attempts (context=mock) + ขยับ true_rank น้ำหนักสูง (mock = ตัวขยับหลัก §5.2)
  // → เรียก computeCalibration (P2) ต่อ topic ด้วย weight สูง (เช่น apply attempt ซ้ำ/ลด RD เร็วกว่า practice)
  for (const a of input.answers) {
    await supabase.from('user_attempts').insert({
      user_id: user.id, question_id: a.questionId, correct: a.correct,
      context: 'mock', count_in_calibration: true,
    })
  }
  return { ok: true, score, weakTopics: weak, admitChance: chance }
}
```
